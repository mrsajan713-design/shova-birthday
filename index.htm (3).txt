    <script>
        function reveal(el) {
            el.querySelector('.card-front').style.display = 'none';
            el.querySelector('.card-content').style.display = 'block';
            
            // Audio starts playing on the very first click
            var audio = document.getElementById('music');
            audio.play().catch(function(error) {
                console.log("Audio play failed, waiting for more interaction");
            });
            
            confetti({ particleCount: 100, spread: 70, origin: { y: 0.6 } });
        }

        let state = 0;
        function cakeFlow() {
            const pic = document.getElementById('cake-pic');
            const btn = document.getElementById('cake-btn');
            const txt = document.getElementById('cake-text');

            if(state === 0) {
                txt.innerText = "Now cut the cake!";
                btn.innerText = "Click Cake to Cut It 🔪";
                state = 1;
            } else {
                pic.src = "https://cdn-icons-png.flaticon.com/512/2682/2682444.png";
                txt.innerText = "Enjoy your slice! 😋";
                btn.style.display = "none";
                confetti({ particleCount: 200, spread: 100 });
            }
        }

        function openLetter(el) {
            el.innerHTML = "<h2 style='font-family:Dancing Script'>Dear Shova,</h2><p>You are truly special. I hope this little surprise made you smile. Happy Birthday once more! ✨🥳</p>";
            confetti({ particleCount: 150, spread: 70 });
        }
